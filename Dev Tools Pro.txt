   function toggleMobileMenu() {
    const menu = document.getElementById('mobileMenu');
    const menuBtn = document.getElementById('mobileMenuBtn');
    
    if (menu && menuBtn) {
        const isActive = menu.classList.toggle('active');
        
        // Update menu icon
        const menuIcon = menuBtn.querySelector('i');
        menuIcon.className = isActive ? 'fas fa-times' : 'fas fa-bars';
        
        // Prevent/allow body scrolling
        document.body.style.overflow = isActive ? 'hidden' : '';
        
        // Force reflow to ensure smooth animation
        menu.style.display = 'block';
        setTimeout(() => {
            menu.style.display = isActive ? 'block' : 'none';
        }, 10);
    }
}
    
function closeMobileMenu() {
    const menu = document.getElementById('mobileMenu');
    const menuBtn = document.getElementById('mobileMenuBtn');
    
    if (menu && menuBtn) {
        menu.classList.remove('active');
        menu.style.display = 'none';
        
        const menuIcon = menuBtn.querySelector('i');
        menuIcon.className = 'fas fa-bars';
        
        document.body.style.overflow = '';
    }
}
      
    // Close mobile menu when clicking outside
    document.addEventListener('click', function(event) {
        const menu = document.getElementById('mobileMenu');
        const menuBtn = document.getElementById('mobileMenuBtn');
        
        if (menu && menu.classList.contains('active') && 
            !menu.contains(event.target) && 
            !menuBtn.contains(event.target)) {
            closeMobileMenu();
        }
    });
    
    // Close mobile menu when window is resized to desktop
    window.addEventListener('resize', function() {
        if (window.innerWidth >= 769) {
            closeMobileMenu();
        }
    });

function handleImageUpload(event) {
    const files = event.target.files;
    if (!files.length) return;
    
    for (let i = 0; i < files.length; i++) {
        const file = files[i];
        
        // Check file size (10MB limit)
        if (file.size > 10 * 1024 * 1024) {
            showCustomNotification('Error', `File "${file.name}" is too large (max 10MB)`, 'danger');
            continue;
        }
        
        // Check file type
        if (!file.type.startsWith('image/')) {
            showCustomNotification('Error', `File "${file.name}" is not an image`, 'danger');
            continue;
        }
        
        imageFiles.push({
            id: Date.now() + i,
            file: file,
            name: file.name,
            size: file.size,
            url: URL.createObjectURL(file),
            optimized: null
        });
    }
    
    updateBatchList();
    if (imageFiles.length > 0) {
        document.getElementById('batchProcessing').style.display = 'block';
        previewFirstImage();
    }
}

}

function formatFileSize(bytes) {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
}

function optimizeSingleImage(imageData) {
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    const img = new Image();
    
    img.onload = function() {
        // Get resize dimensions
        let width = parseInt(document.getElementById('resizeWidth').value) || this.naturalWidth;
        let height = parseInt(document.getElementById('resizeHeight').value) || this.naturalHeight;
        
        // Keep aspect ratio if checked
        if (document.getElementById('keepAspectRatio').checked) {
            const ratio = Math.min(width / this.naturalWidth, height / this.naturalHeight);
            width = Math.round(this.naturalWidth * ratio);
            height = Math.round(this.naturalHeight * ratio);
        }
        
        // Set canvas dimensions
        canvas.width = width;
        canvas.height = height;
        
        // Draw image on canvas
        ctx.drawImage(img, 0, 0, width, height);
        
    // Define MIME type based on selected format
    let mimeType = imageData.file.type;
    switch(currentFormat) {
        case 'jpg':
        case 'jpeg':
            mimeType = 'image/jpeg';
            break;
        case 'png':
            mimeType = 'image/png';
            break;
        case 'webp':
            mimeType = 'image/webp';
            break;
    }
    
    // Get quality (convert 0-100 to 0-1 for WebP)
    let quality = currentQuality / 100;
    if (mimeType === 'image/webp') {
        quality = Math.min(0.9, quality); // WebP quality range is different
    }
    
    // Convert to blob
    canvas.toBlob(function(blob) {
        if (!blob) {
            showCustomNotification('Error', 'Failed to optimize image', 'danger');
            return;
        }
        
        const optimizedUrl = URL.createObjectURL(blob);
        
        // Update imageData
        imageData.optimized = {
            blob: blob,
            url: optimizedUrl,
            size: blob.size,
            width: width,
            height: height
        };
        
        // Update preview
        const optimizedPreview = document.getElementById('optimizedPreview');
        const optimizedStats = document.getElementById('optimizedStats');
        
        if (optimizedPreview) {
            optimizedPreview.src = optimizedUrl;
            optimizedStats.innerHTML = `
                <div class="image-stat-item">
                    <div class="image-stat-value">${width} × ${height}</div>
                    <div class="image-stat-label">Dimensions</div>
                </div>
                <div class="image-stat-item">
                    <div class="image-stat-value">${formatFileSize(blob.size)}</div>
                    <div class="image-stat-label">File Size</div>
                </div>
                <div class="image-stat-item">
                    <div class="image-stat-value">${Math.round((1 - blob.size / imageData.size) * 100)}%</div>
                    <div class="image-stat-label">Reduction</div>
                </div>
            `;
            
            // Update savings percentage
            const savings = Math.round((1 - blob.size / imageData.size) * 100);
            document.getElementById('savingsPercent').textContent = savings + '%';
            
            // Update optimized images list
            updateOptimizedList();
        }
    }, mimeType, quality);
};

img.src = imageData.url;
}

function createChart() {
    const ctx = document.getElementById('chartCanvas').getContext('2d');
    
    if (chart) {
        chart.destroy();
    }
    
    const options = {
        responsive: true,
        maintainAspectRatio: false,
        plugins: {
            legend: {
                display: document.getElementById('showLegend').checked,
                position: 'top'
            },
            tooltip: {
                enabled: document.getElementById('showTooltips').checked
            },
            title: {
                display: true,
                text: document.getElementById('chartTitle').value || 'Chart',
                font: {
                    size: 16
                }
            }
        },
        scales: currentChartType === 'bar' || currentChartType === 'line' ? {
            x: {
                title: {
                    display: !!document.getElementById('xAxisLabel').value,
                    text: document.getElementById('xAxisLabel').value
                },
                grid: {
                    display: document.getElementById('showGrid').checked
                }
            },
            y: {
                title: {
                    display: !!document.getElementById('yAxisLabel').value,
                    text: document.getElementById('yAxisLabel').value
                },
                grid: {
                    display: document.getElementById('showGrid').checked
                },
                beginAtZero: true
            }
        } : {},
        animation: document.getElementById('animateChart').checked ? {
            duration: 1000,
            easing: 'easeOutQuart'
        } : {}
    };
    
    // Special options for different chart types
    if (currentChartType === 'pie' || currentChartType === 'doughnut') {
        options.plugins.legend.position = 'right';
    }
    
    if (currentChartType === 'radar') {
        options.scales = {
            r: {
                beginAtZero: true,
                grid: {
                    display: document.getElementById('showGrid').checked
                }
            }
        };
    }
    
    chart = new Chart(ctx, {
        type: currentChartType,
        data: currentData,
        options: options
    });
    
    updateChartStats();
}

function updateChart() {
    if (!chart) return;
    
    // Update chart options
    chart.options.plugins.legend.display = document.getElementById('showLegend').checked;
    chart.options.plugins.tooltip.enabled = document.getElementById('showTooltips').checked;
    chart.options.plugins.title.text = document.getElementById('chartTitle').value || 'Chart';
    
    // Update animation
    chart.options.animation = document.getElementById('animateChart').checked ? {
        duration: 1000,
        easing: 'easeOutQuart'
    } : {};
    
    // Update grid visibility
    if (chart.options.scales) {
        if (chart.options.scales.x) {
            chart.options.scales.x.grid.display = document.getElementById('showGrid').checked;
            chart.options.scales.x.title = {
                display: !!document.getElementById('xAxisLabel').value,
                text: document.getElementById('xAxisLabel').value
            };
        }
        if (chart.options.scales.y) {
            chart.options.scales.y.grid.display = document.getElementById('showGrid').checked;
            chart.options.scales.y.title = {
                display: !!document.getElementById('yAxisLabel').value,
                text: document.getElementById('yAxisLabel').value
            };
        }
        if (chart.options.scales.r) {
            chart.options.scales.r.grid.display = document.getElementById('showGrid').checked;
        }
    }
    
    chart.update();
}

function addDataColumn() {
    const headerRow = document.getElementById('tableHeaders');
    const bodyRows = document.querySelectorAll('#tableBody tr');
    const colCount = headerRow.children.length;
    
    // Add header
    const newHeader = document.createElement('th');
    newHeader.textContent = `Value ${colCount}`;
    headerRow.appendChild(newHeader);
    
    // Add cells to each row
    bodyRows.forEach((row, rowIndex) => {
        const newCell = document.createElement('td');
        const input = document.createElement('input');
        input.type = 'number';
        input.value = Math.floor(Math.random() * 100) + 1;
        input.setAttribute('data-row', rowIndex);
        input.setAttribute('data-col', colCount - 1);
        input.onchange = updateChartData;
        newCell.appendChild(input);
        row.appendChild(newCell);
    });
    
    updateChartData();

function addDataRow() {
    const tableBody = document.getElementById('tableBody');
    const rowCount = tableBody.children.length;
    const colCount = document.getElementById('tableHeaders').children.length;
    
    const newRow = document.createElement('tr');
    
    for (let i = 0; i < colCount; i++) {
        const cell = document.createElement('td');
        const input = document.createElement('input');
        
        if (i === 0) {
            input.type = 'text';
            input.value = `Month ${rowCount + 1}`;
        } else {
            input.type = 'number';
            input.value = Math.floor(Math.random() * 100) + 1;
        }
        
        input.setAttribute('data-row', rowCount);
        input.setAttribute('data-col', i);
        input.onchange = updateChartData;
        cell.appendChild(input);
        newRow.appendChild(cell);
    }
    
    tableBody.appendChild(newRow);
    updateChartData();
}

function clearDataTable() {
    const tableBody = document.getElementById('tableBody');
    const inputs = tableBody.querySelectorAll('input');
    
    inputs.forEach(input => {
        if (input.getAttribute('data-col') === '0') {
            input.value = '';
        } else {
            input.value = '0';
        }
    });
    
    updateChartData();
}

function updateChartData() {
    const labels = [];
    const datasets = [];
    const tableBody = document.getElementById('tableBody');
    const rows = tableBody.querySelectorAll('tr');
    
    if (rows.length === 0) return;
    
    // Get column count from first row
    const firstRowInputs = rows[0].querySelectorAll('input');
    const colCount = firstRowInputs.length;
    
    // Extract labels (first column)
    rows.forEach(row => {
        const labelInput = row.querySelector('input[data-col="0"]');
        if (labelInput && labelInput.value) {
            labels.push(labelInput.value);
        } else {
            labels.push('');
        }
    });
    
    // Extract datasets (other columns)
    for (let col = 1; col < colCount; col++) {
        const dataset = {
            label: `Dataset ${col}`,
            data: [],
            backgroundColor: colorPalettes.default.background[(col - 1) % colorPalettes.default.background.length],
            borderColor: colorPalettes.default.border[(col - 1) % colorPalettes.default.border.length],
            borderWidth: 2
        };
        
        rows.forEach(row => {
            const input = row.querySelector(`input[data-col="${col}"]`);
            if (input) {
                dataset.data.push(parseFloat(input.value) || 0);
            }
        });
        
        datasets.push(dataset);
    }
    
    // Update current data
    currentData.labels = labels;
    currentData.datasets = datasets;
    
    // Update chart
    if (chart) {
        chart.data = currentData;
        chart.update();
    } else {
        createChart();
    }
    
    updateChartStats();
}

function updateTableFromData(labels, datasets) {
    const tableBody = document.getElementById('tableBody');
    tableBody.innerHTML = '';
    
    // Clear existing headers except first
    const headerRow = document.getElementById('tableHeaders');
    while (headerRow.children.length > 1) {
        headerRow.removeChild(headerRow.lastChild);
    }
    
    // Add new headers
    datasets.forEach((dataset, index) => {
        const th = document.createElement('th');
        th.textContent = dataset.label || `Value ${index + 1}`;
        headerRow.appendChild(th);
    });
    
    // Add rows
    labels.forEach((label, rowIndex) => {
        const tr = document.createElement('tr');
        
        // Label cell
        const labelTd = document.createElement('td');
        const labelInput = document.createElement('input');
        labelInput.type = 'text';
        labelInput.value = label;
        labelInput.setAttribute('data-row', rowIndex);
        labelInput.setAttribute('data-col', 0);
        labelInput.onchange = updateChartData;
        labelTd.appendChild(labelInput);
        tr.appendChild(labelTd);
        
        // Data cells
        datasets.forEach((dataset, colIndex) => {
            const td = document.createElement('td');
            const input = document.createElement('input');
            input.type = 'number';
            input.value = dataset.data[rowIndex] || 0;
            input.setAttribute('data-row', rowIndex);
            input.setAttribute('data-col', colIndex + 1);
            input.onchange = updateChartData;
            td.appendChild(input);
            tr.appendChild(td);
        });
        
        tableBody.appendChild(tr);
    });
}

function importCSVData() {
    const csvText = document.getElementById('csvInput').value.trim();
    if (!csvText) {
        showCustomNotification('Error', 'Please enter CSV data', 'warning');
        return;
    }
    
    try {
        const lines = csvText.split('\n').map(line => line.trim()).filter(line => line);
        if (lines.length < 2) {
            throw new Error('CSV must have at least 2 lines (header and data)');
        }
        
        const headers = lines[0].split(',').map(h => h.trim());
        const labels = [];
        const datasets = [];
        
        // Initialize datasets
        for (let i = 1; i < headers.length; i++) {
            datasets.push({
                label: headers[i],
                data: [],
                backgroundColor: colorPalettes.default.background[(i - 1) % colorPalettes.default.background.length],
                borderColor: colorPalettes.default.border[(i - 1) % colorPalettes.default.border.length],
                borderWidth: 2
            });
        }
        
        // Parse data rows
        for (let i = 1; i < lines.length; i++) {
            const values = lines[i].split(',').map(v => v.trim());
            labels.push(values[0]); // First column is labels
            
            for (let j = 1; j < values.length && j - 1 < datasets.length; j++) {
                datasets[j - 1].data.push(parseFloat(values[j]) || 0);
            }
        }
        
        // Update table
        updateTableFromData(labels, datasets);
        
        // Update chart
        currentData.labels = labels;
        currentData.datasets = datasets;
        
        if (chart) {
            chart.data = currentData;
            chart.update();
        } else {
            createChart();
        }
        
        updateChartStats();
        
        showCustomNotification('Success', 'CSV data imported successfully', 'success');
        
    } catch (error) {
        showCustomNotification('Error', 'Invalid CSV format: ' + error.message, 'danger');
    }
}

function importJSONData() {
    const jsonText = document.getElementById('jsonInput').value.trim();
    if (!jsonText) {
        showCustomNotification('Error', 'Please enter JSON data', 'warning');
        return;
    }
    
    try {
        const data = JSON.parse(jsonText);
        
        if (!data.labels || !data.datasets) {
            throw new Error('JSON must have "labels" and "datasets" properties');
        }
        
        currentData = data;
        
        // Update table from JSON
        updateTableFromData(data.labels, data.datasets);
        
        // Update chart
        if (chart) {
            chart.data = currentData;
            chart.update();
        } else {
            createChart();
        }
        
        updateChartStats();
        
        showCustomNotification('Success', 'JSON data imported successfully', 'success');
        
    } catch (error) {
        showCustomNotification('Error', 'Invalid JSON: ' + error.message, 'danger');
    }
}

function generateRandomData() {
    const dataPoints = parseInt(document.getElementById('dataPoints').value) || 6;
    const datasetsCount = parseInt(document.getElementById('dataSets').value) || 2;
    const minValue = parseInt(document.getElementById('minValue').value) || 10;
    const maxValue = parseInt(document.getElementById('maxValue').value) || 100;
    
    const labels = [];
    for (let i = 1; i <= dataPoints; i++) {
        labels.push(`Data ${i}`);
    }
    
    const datasets = [];
    for (let i = 0; i < datasetsCount; i++) {
        const data = [];
        for (let j = 0; j < dataPoints; j++) {
            data.push(Math.floor(Math.random() * (maxValue - minValue + 1)) + minValue);
        }
        
        datasets.push({
            label: `Dataset ${i + 1}`,
            data: data,
            backgroundColor: colorPalettes.default.background[i % colorPalettes.default.background.length],
            borderColor: colorPalettes.default.border[i % colorPalettes.default.border.length],
            borderWidth: 2
        });
    }
    
    // Update table
    updateTableFromData(labels, datasets);
    
    // Update chart
    currentData.labels = labels;
    currentData.datasets = datasets;
    
    if (chart) {
        chart.data = currentData;
        chart.update();
    } else {
        createChart();
    }
    
    updateChartStats();
    
    showCustomNotification('Random Data Generated', `${datasetsCount} datasets with ${dataPoints} points each`, 'success');
}

    function adjustColor(color, amount) {
        let usePound = false;
        if (color[0] === "#") {
            color = color.slice(1);
            usePound = true;
        }
        const num = parseInt(color, 16);
        let r = (num >> 16) + amount;
        if (r > 255) r = 255;
        else if (r < 0) r = 0;
        let b = ((num >> 8) & 0x00FF) + amount;
        if (b > 255) b = 255;
        else if (b < 0) b = 0;
        let g = (num & 0x0000FF) + amount;
        if (g > 255) g = 255;
        else if (g < 0) g = 0;
        return (usePound ? "#" : "") + (g | (b << 8) | (r << 16)).toString(16).padStart(6, '0');
    }

function exportChartAsImage(format = 'png') {
    if (!chart) {
        showCustomNotification('Error', 'No chart to export', 'warning');
        return;
    }
    
    const canvas = document.getElementById('chartCanvas');
    if (!canvas) {
        showCustomNotification('Error', 'Chart canvas not found', 'warning');
        return;
    }
    
    // Set image quality and type
    let mimeType, fileName;
    
    if (format === 'jpeg' || format === 'jpg') {
        mimeType = 'image/jpeg';
        fileName = `chart-${Date.now()}.jpg`;
    } else if (format === 'png') {
        mimeType = 'image/png';
        fileName = `chart-${Date.now()}.png`;
    } else {
        mimeType = 'image/png';
        fileName = `chart-${Date.now()}.png`;
    }
    
    // Convert canvas to data URL
    const dataURL = canvas.toDataURL(mimeType, 1.0);
    
    // Create download link
    const link = document.createElement('a');
    link.href = dataURL;
    link.download = fileName;
    
    // Trigger download
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    
    showCustomNotification('Success', `Chart exported as ${format.toUpperCase()}`, 'success');
}

    function encodeUrl(url) {
        return btoa(encodeURIComponent(url)).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }

    function decodeUrl(encoded) {
        try {
            while (encoded.length % 4) {
                encoded += '=';
            }
            encoded = encoded.replace(/-/g, '+').replace(/_/g, '/');
            return decodeURIComponent(atob(encoded));
        } catch (e) {
            return null;
        }
    }

    function checkForRedirect() {
        const urlParams = new URLSearchParams(window.location.search);
        const urlParam = urlParams.get('u');
        const timeParam = urlParams.get('t');
        
        if (urlParam) {
            isOnRedirectPage = true;
            showRedirectPage();
            
            const decodedUrl = decodeUrl(urlParam);
            
            if (decodedUrl && isValidUrl(decodedUrl)) {
                currentRedirectUrl = decodedUrl;
                currentRedirectTime = timeParam ? parseInt(timeParam) : 10;
                
                shortenerStats.clicks++;
                saveShortenerStats();
                initShortenerStats();
                
                startRedirectCountdown();
            } else {
                showInvalidLink();
            }
        } else {
            isOnRedirectPage = false;
        }
    }

    function startRedirectCountdown() {
        let countdown = currentRedirectTime;
        const countdownNumber = document.getElementById('redirectCountdownNumber');
        const countdownSeconds = document.getElementById('redirectCountdownSeconds');
        const progressBar = document.getElementById('redirectProgressBar');
        
        if (!countdownNumber || !countdownSeconds || !progressBar) return;
        
        countdownNumber.textContent = countdown;
        countdownSeconds.textContent = countdown;
        progressBar.style.width = '100%';
        
        const countdownBox = document.getElementById('redirectCountdownBox');
        if (countdownBox) {
            countdownBox.style.display = 'block';
        }
        
        redirectInterval = setInterval(() => {
            countdown--;
            
            countdownNumber.textContent = countdown;
            countdownSeconds.textContent = countdown;
            
            const progress = (countdown / currentRedirectTime) * 100;
            progressBar.style.width = progress + '%';
            
            if (countdown <= 0) {
                clearInterval(redirectInterval);
                performRedirect();
            }
        }, 1000);
    }

    function skipRedirectCountdown() {
        if (redirectInterval) {
            clearInterval(redirectInterval);
        }
        performRedirect();
    }

    function generateShortCode(length = 6) {
        const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
        let result = '';
        for (let i = 0; i < length; i++) {
            result += chars.charAt(Math.floor(Math.random() * chars.length));
        }
        return result;
    }

    function shortenUrlNew() {
        const originalUrl = document.getElementById('originalUrl')?.value.trim();
        const customSlug = document.getElementById('customSlug')?.value.trim();
        const countdownTime = parseInt(document.getElementById('countdownTime')?.value || 10);
        const enableEncryption = document.getElementById('enableEncryption')?.checked || true;
        
        if (!originalUrl) {
            showCustomNotification('Missing URL', 'Please enter a URL to shorten.', 'warning');
            return;
        }
        
        if (!isValidUrl(originalUrl)) {
            showCustomNotification('Invalid URL', 'Please enter a valid URL (include http:// or https://).', 'warning');
            return;
        }
        
        let shortCode;
        if (customSlug) {
            shortCode = customSlug.replace(/[^a-zA-Z0-9-_]/g, '').toLowerCase();
            if (!shortCode) {
                showCustomNotification('Invalid Slug', 'Custom slug can only contain letters, numbers, hyphens, and underscores.', 'warning');
                return;
            }
        } else {
            shortCode = generateShortCode();
        }
        
        const encodedUrl = enableEncryption ? encodeUrl(originalUrl) : encodeURIComponent(originalUrl);
        const baseUrl = window.location.origin + window.location.pathname;
        const shortUrl = `${baseUrl}?u=${encodedUrl}&t=${countdownTime}${customSlug ? `&c=${shortCode}` : ''}`;
        
        const urlData = {
            id: Date.now(),
            shortCode: shortCode,
            originalUrl: originalUrl,
            shortUrl: shortUrl,
            countdownTime: countdownTime,
            clicks: 0,
            created: new Date().toISOString(),
            createdDate: new Date().toDateString()
        };
        
        urlDatabase.unshift(urlData);
        
        shortenerStats.total++;
        shortenerStats.today++;
        saveShortenerStats();
        saveUrlDatabase();
        
        const shortUrlLink = document.getElementById('shortUrlLinkNew');
        const resultSection = document.getElementById('resultSectionNew');
        const countdownPreview = document.getElementById('countdownPreviewNew');
        
        if (shortUrlLink) shortUrlLink.textContent = shortUrl;
        if (shortUrlLink) shortUrlLink.href = shortUrl;
        if (countdownPreview) countdownPreview.textContent = countdownTime;
        if (resultSection) resultSection.classList.add('active');
        
        generateQRCodeNew(shortUrl);
        
        if (resultSection) {
            resultSection.scrollIntoView({ behavior: 'smooth' });
        }
        
        initShortenerStats();
        
        showCustomNotification('URL Shortened', 'Your short URL has been generated successfully!', 'success');
    }

    function obfuscateJavaScript() {
        const jsCode = document.getElementById('jsInput').value;
        if (!jsCode.trim()) {

            return;
        }
        
        // Get settings
        const compactCode = document.getElementById('jsCompactCode').checked;
        const selfDefending = document.getElementById('jsSelfDefending').checked;

        const stringArrayEncoding = document.getElementById('jsStringArrayEncoding').value;
        const identifierPrefix = document.getElementById('jsIdentifierPrefix').value;
        
        let obfuscatedCode = jsCode;
        
        // Basic obfuscation (for demo purposes)
        // In a real implementation, you would use a proper obfuscation library
        
        // String array encoding (simplified)
        if (stringArray) {
            const strings = [];
            const stringRegex = /(["'])(?:(?=(\\?))\2.)*?\1/g;
            let match;
            
            while ((match = stringRegex.exec(jsCode)) !== null) {
                strings.push(match[0]);
            }
            
            // Create string array
            if (strings.length > 0) {
                let stringArrayName = '_0x' + Math.random().toString(16).substr(2, 8);
                let stringArrayCode = `var ${stringArrayName}=[${strings.join(',')}];\n`;
                
                // Replace strings with array references
                obfuscatedCode = stringArrayCode + obfuscatedCode;
                strings.forEach((str, index) => {
                    const regex = new RegExp(str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
                    obfuscatedCode = obfuscatedCode.replace(regex, `${stringArrayName}[${index}]`);
                });
            }
        }
        
        // Identifier renaming (basic)
        if (identifierNames) {
            const identifierRegex = /\b(var|let|const)\s+([a-zA-Z_$][a-zA-Z0-9_$]*)\b/g;
            const identifiers = new Map();
            let counter = 0;
            
            // Find all identifiers
            let match;
            while ((match = identifierRegex.exec(obfuscatedCode)) !== null) {
                const identifier = match[2];
                if (!identifiers.has(identifier) && !identifier.startsWith('_0x')) {
                    identifiers.set(identifier, identifierPrefix + 'x' + counter++);
                }
            }
            
            // Replace identifiers
            for (const [original, obfuscated] of identifiers) {

    function generatePassword() {
        const length = parseInt(document.getElementById('passwordLength').value);
        const includeUppercase = document.getElementById('includeUppercase').checked;
        const includeLowercase = document.getElementById('includeLowercase').checked;
        const includeNumbers = document.getElementById('includeNumbers').checked;
        const includeSymbols = document.getElementById('includeSymbols').checked;
        const excludeChars = document.getElementById('excludeChars').value;
        
        let charset = '';
        if (includeUppercase) charset += 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
        if (includeLowercase) charset += 'abcdefghijklmnopqrstuvwxyz';
        if (includeNumbers) charset += '0123456789';
        if (includeSymbols) charset += '!@#$%^&*()_+-=[]{}|;:,.<>?';
        
        if (charset === '') {
            showCustomNotification('Error', 'Please select at least one character type.', 'warning');
            return;
        }
        
        // Remove excluded characters
        if (excludeChars) {
            const excludeRegex = new RegExp(`[${excludeChars.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}]`, 'g');
            charset = charset.replace(excludeRegex, '');
        }
        
        if (charset.length === 0) {
            showCustomNotification('Error', 'No characters available after exclusions.', 'warning');
            return;
        }
        
        let password = '';
        const array = new Uint32Array(length);
        window.crypto.getRandomValues(array);
        
        for (let i = 0; i < length; i++) {
            password += charset[array[i] % charset.length];
        }
        
        currentPassword = password;
        
        // Update UI
        document.getElementById('passwordPreview').textContent = password;
        document.getElementById('passwordResult').classList.add('active');
        
        // Calculate strength
        const strength = calculatePasswordStrength(password);
        const strengthBar = document.getElementById('strengthBar');
        const strengthText = document.getElementById('strengthText');
        
        strengthBar.className = 'strength-bar ' + strength.class;
        strengthBar.style.width = strength.percentage + '%';
        strengthText.textContent = strength.text;
        strengthText.style.color = strength.color;
        
        // Update stats
        passwordStats.total++;
        passwordStats.today++;
        localStorage.setItem('MyStudy_password_stats', JSON.stringify(passwordStats));
        updateDashboardStats();
        
        showCustomNotification('Password Generated', 'Strong password created successfully!', 'success');
    }

    function calculatePasswordStrength(password) {
        let score = 0;
        const length = password.length;
        
        // Length score
        if (length >= 8) score += 1;
        if (length >= 12) score += 1;
        if (length >= 16) score += 2;
        
        // Character variety score
        if (/[A-Z]/.test(password)) score += 1;
        if (/[a-z]/.test(password)) score += 1;
        if (/[0-9]/.test(password)) score += 1;
        if (/[^A-Za-z0-9]/.test(password)) score += 2;
        
        // Check for common patterns
        if (/(.)\1{2,}/.test(password)) score -= 1; // Repeated characters
        if (/123|abc|qwerty|password/i.test(password)) score -= 2; // Common patterns
        
        // Calculate percentage
        let percentage = Math.min(100, Math.max(0, (score / 8) * 100));
        
        // Determine strength class
        let strengthClass, strengthText, color;
        if (percentage < 30) {
            strengthClass = 'strength-weak';
            strengthText = 'Weak';
            color = 'var(--danger)';
        } else if (percentage < 60) {
            strengthClass = 'strength-medium';
            strengthText = 'Medium';
            color = 'var(--warning)';
        } else if (percentage < 85) {
            strengthClass = 'strength-strong';
            strengthText = 'Strong';
            color = 'var(--success)';
        } else {
            strengthClass = 'strength-very-strong';
            strengthText = 'Very Strong';
            color = 'var(--primary)';
        }
        
        return {
            percentage: percentage,
            class: strengthClass,
            text: strengthText,
            color: color
        };
    }

    function caesarCipher(text, shift) {
        return text.replace(/[a-z]/gi, function(char) {
            const base = char <= 'Z' ? 65 : 97;
            return String.fromCharCode((char.charCodeAt(0) - base + shift) % 26 + base);
        });
    }

    function xorCipher(text, key) {
        let result = '';
        for (let i = 0; i < text.length; i++) {
            result += String.fromCharCode(text.charCodeAt(i) ^ key.charCodeAt(i % key.length));
        }
        return btoa(result);
    }

    function atbashCipher(text) {
        return text.replace(/[a-z]/gi, function(char) {
            const base = char <= 'Z' ? 65 : 97;
            return String.fromCharCode(25 - (char.charCodeAt(0) - base) + base);
        });
    }

            const className = match[1];
            if (!classMap.has(className)) {
                classMap.set(className, prefix + generateObfuscatedName(counter++));
            }
        }
        
        // Find all IDs
        const idRegex = /#([a-zA-Z_][a-zA-Z0-9_-]*)/g;
        while ((match = idRegex.exec(css)) !== null) {
            const idName = match[1];
            if (!idMap.has(idName)) {
                idMap.set(idName, prefix + generateObfuscatedName(counter++));
            }
        }
        
        // Replace class names and IDs
        let obfuscatedCSS = css;
        
        // Replace classes
        for (const [original, obfuscated] of classMap) {
            const regex = new RegExp(`\\.${original}(?![a-zA-Z0-9_-])`, 'g');
            obfuscatedCSS = obfuscatedCSS.replace(regex, '.' + obfuscated);
        }
        
        // Replace IDs
        for (const [original, obfuscated] of idMap) {
            const regex = new RegExp(`#${original}(?![a-zA-Z0-9_-])`, 'g');
            obfuscatedCSS = obfuscatedCSS.replace(regex, '#' + obfuscated);
        }
        
        // Minify if requested
        if (minify) {
            obfuscatedCSS = obfuscatedCSS
                .replace(/\/\*[\s\S]*?\*\//g, '') // Remove comments
                .replace(/\s+/g, ' ') // Replace multiple spaces with single space
                .replace(/\s*\{\s*/g, '{')
                .replace(/\s*\}\s*/g, '}')
                .replace(/\s*;\s*/g, ';')
                .replace(/\s*:\s*/g, ':')
                .replace(/\s*,\s*/g, ',')
                .trim();
        }
        
        // Update UI
        document.getElementById('obfuscatedCSSOutput').textContent = obfuscatedCSS;
        document.getElementById('cssObfuscatorResult').classList.add('active');
        
        // Calculate stats
        const originalSize = new Blob([css]).size;
        const obfuscatedSize = new Blob([obfuscatedCSS]).size;
        const reduction = originalSize > 0 ? Math.round((1 - obfuscatedSize / originalSize) * 100) : 0;
        
        document.getElementById('originalSize').textContent = originalSize;
        document.getElementById('obfuscatedSize').textContent = obfuscatedSize;
        document.getElementById('reductionPercent').textContent = reduction + '%';
        document.getElementById('cssStats').style.display = 'flex';
        
        showCustomNotification('CSS Obfuscated', `CSS obfuscated with ${classMap.size + idMap.size} renamed classes/IDs`, 'success');
    }

    function generateObfuscatedName(index) {
        const chars = 'abcdefghijklmnopqrstuvwxyz';
        let name = '';
        let i = index;
        
        do {
            name = chars[i % chars.length] + name;
            i = Math.floor(i / chars.length) - 1;
        } while (i >= 0);
        
        return name;
    }

    function minifyCSS() {
        const css = document.getElementById('minifyInput').value;
        if (!css.trim()) {
            showCustomNotification('Error', 'Please enter CSS code.', 'warning');
            return;
        }
        
        let minified = css;
        
        // Remove comments
        if (document.getElementById('removeComments').checked) {
            minified = minified.replace(/\/\*[\s\S]*?\*\//g, '');
        }
        
        // Remove whitespace
        if (document.getElementById('removeWhitespace').checked) {
            minified = minified
                .replace(/\s+/g, ' ')
                .replace(/\s*\{\s*/g, '{')
                .replace(/\s*\}\s*/g, '}')
                .replace(/\s*;\s*/g, ';')
                .replace(/\s*:\s*/g, ':')
                .replace(/\s*,\s*/g, ',');
        }
        
        // Shorten colors
        if (document.getElementById('shortenColors').checked) {
            minified = minified.replace(/#([a-f0-9])\1([a-f0-9])\2([a-f0-9])\3/gi, '#$1$2$3');
        }
        
        // Shorten zero values
        if (document.getElementById('shortenZeroValues').checked) {
            minified = minified.replace(/(\s|:)0(px|em|rem|%|vh|vw)/g, '$10');
        }
        
        // Remove last semicolons
        if (document.getElementById('removeSemicolons').checked) {
            minified = minified.replace(/;\s*}/g, '}');
        }
        
        // Trim
        minified = minified.trim();
        
        // Update UI
        document.getElementById('minifiedOutput').textContent = minified;
        document.getElementById('minifierResult').classList.add('active');
        
        // Calculate stats
        const originalSize = new Blob([css]).size;
        const minifiedSize = new Blob([minified]).size;
        const reduction = originalSize > 0 ? Math.round((1 - minifiedSize / originalSize) * 100) : 0;
        
        document.getElementById('originalCSSSize').textContent = originalSize;
        document.getElementById('minifiedCSSSize').textContent = minifiedSize;
        document.getElementById('cssReductionPercent').textContent = reduction + '%';
        
        showCustomNotification('CSS Minified', `Size reduced by ${reduction}%`, 'success');
    }

    function formatCSS() {
        const css = document.getElementById('minifyInput').value;
        if (!css.trim()) {
            showCustomNotification('Error', 'Please enter CSS code.', 'warning');
            return;
        }
        
        // Basic formatting
        let formatted = css
            .replace(/\s*\{\s*/g, ' {\n  ')
            .replace(/\s*;\s*/g, ';\n  ')
            .replace(/\s*\}\s*/g, '\n}\n')
            .replace(/\s*,\s*/g, ', ')
            .replace(/\s+/g, ' ');
        
        // Indent properly
        let indentLevel = 0;
        const lines = formatted.split('\n');
        formatted = lines.map(line => {
            line = line.trim();
            if (line.includes('}')) indentLevel = Math.max(0, indentLevel - 1);
            const indented = '  '.repeat(indentLevel) + line;
            if (line.includes('{')) indentLevel++;
            return indented;
        }).join('\n');
        
        document.getElementById('minifyInput').value = formatted.trim();
        showCustomNotification('CSS Formatted', 'CSS code has been formatted.', 'info');
    }

                    }[m]));

                    const textarea = document.createElement('textarea');

                    break;
            }
            
            document.getElementById('encoderOutput').textContent = output;
            document.getElementById('encoderResult').classList.add('active');
            showCustomNotification('Success', 'Text processed successfully!', 'success');

    function hexToRGB(hex) {
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return { r, g, b };
    }
    
    function RGBToHSL(r, g, b) {
        r /= 255;
        g /= 255;
        b /= 255;
        
        const max = Math.max(r, g, b);
        const min = Math.min(r, g, b);
        let h, s, l = (max + min) / 2;
        
        if (max === min) {
            h = s = 0;
        } else {
            const d = max - min;
            s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
            switch(max) {
                case r: h = (g - b) / d + (g < b ? 6 : 0); break;
                case g: h = (b - r) / d + 2; break;
                case b: h = (r - g) / d + 4; break;
            }
            h /= 6;
        }
        
        return {
            h: Math.round(h * 360),
            s: Math.round(s * 100),
            l: Math.round(l * 100)
        };
    }
    
    function HSLToHex(h, s, l) {
        s /= 100;
        l /= 100;
        
        const c = (1 - Math.abs(2 * l - 1)) * s;
        const x = c * (1 - Math.abs((h / 60) % 2 - 1));
        const m = l - c / 2;
        
        let r, g, b;
        
        if (h >= 0 && h < 60) {
            [r, g, b] = [c, x, 0];
        } else if (h >= 60 && h < 120) {
            [r, g, b] = [x, c, 0];
        } else if (h >= 120 && h < 180) {
            [r, g, b] = [0, c, x];
        } else if (h >= 180 && h < 240) {
            [r, g, b] = [0, x, c];
        } else if (h >= 240 && h < 300) {
            [r, g, b] = [x, 0, c];
        } else {
            [r, g, b] = [c, 0, x];
        }
        
        r = Math.round((r + m) * 255);
        g = Math.round((g + m) * 255);
        b = Math.round((b + m) * 255);
        
        return `#${((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1)}`;
    }
    
    function generateColorScheme() {
        const hex = document.getElementById('hexColor').value;
        const type = document.getElementById('schemeType').value;
        const rgb = hexToRGB(hex);
        const hsl = RGBToHSL(rgb.r, rgb.g, rgb.b);
        
        let colors = [];
        
        switch(type) {
            case 'monochromatic':
                for (let i = -2; i <= 2; i++) {
                    const newHsl = { ...hsl, l: Math.max(0, Math.min(100, hsl.l + (i * 15))) };
                    colors.push(HSLToHex(hsl.h, hsl.s, newHsl.l));
                }
                break;
            case 'analogous':
                for (let i = -2; i <= 2; i++) {
                    const newHsl = { ...hsl, h: (hsl.h + (i * 30) + 360) % 360 };
                    colors.push(HSLToHex(newHsl.h, hsl.s, hsl.l));
                }
                break;
            case 'complementary':
                colors = [hex, HSLToHex((hsl.h + 180) % 360, hsl.s, hsl.l)];
                colors.push(HSLToHex(hsl.h, hsl.s, Math.min(100, hsl.l + 20)));
                colors.push(HSLToHex(hsl.h, hsl.s, Math.max(0, hsl.l - 20)));
                colors.push(HSLToHex((hsl.h + 180) % 360, hsl.s, Math.min(100, hsl.l + 20)));
                break;
            case 'triadic':
                colors = [
                    hex,

                    hex,
                    HSLToHex((hsl.h + 90) % 360, hsl.s, hsl.l),
                    HSLToHex((hsl.h + 180) % 360, hsl.s, hsl.l),

    function updateTickerPreview() {
        const text = document.getElementById('tickerText').value;
        const items = text.split('\n').filter(item => item.trim()).length > 0 
            ? text.split('\n').filter(item => item.trim())
            : text.split('|').filter(item => item.trim());
        
        if (items.length === 0) {

        }

        const direction = document.getElementById('tickerDirection').value;
        const fontSize = document.getElementById('tickerFontSize').value;
        const bgColor = document.getElementById('tickerBgColor').value;
