// NAVIGATION MANAGEMENT
// ============================
function showPage(pageId) {
    // Update history
    if (appState.currentPage !== pageId) {
        appState.previousPage = appState.currentPage;
        appState.currentPage = pageId;
        
        // Add to history if not already there
        if (!appState.history.includes(pageId)) {
            appState.history.push(pageId);
        }
        
        saveAppState();
    }
    
    // Hide all pages
    document.getElementById('home').style.display = 'none';
    document.querySelectorAll('.app-container').forEach(container => {
        container.classList.remove('active');
    });
    
    // Update navigation buttons
    document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.classList.remove('active');
    });
    
    // Show selected page
    if (pageId === 'home') {
        document.getElementById('home').style.display = 'block';
        document.querySelector('.nav-btn[onclick="showPage(\'home\')"]').classList.add('active');
    } else {
        document.getElementById(pageId).classList.add('active');
        document.querySelector(`.nav-btn[onclick="showPage('${pageId}')"]`).classList.add('active');
    }
    
    // Initialize the specific app
    if (pageId === 'cgpa') {
        loadCGPAData();
    } else if (pageId === 'exam') {
        initExamApp();
    } else if (pageId === 'lecture') {
        loadLectureData();
    } else if (pageId === 'study') {
        initStudyApp();
    }
}

// Go back to previous page
function goBack() {
    if (appState.history.length > 1) {
        // Remove current page from history
        appState.history.pop();
        // Get previous page
        const previousPage = appState.history[appState.history.length - 1];
        appState.currentPage = previousPage;
        saveAppState();
        showPage(previousPage);
    }
}

// Handle browser back button
window.addEventListener('popstate', function(event) {
    goBack();
});

        }, 100);
    }
}

function calculate() {
    let units = 0, pts = 0;
    const scale = document.getElementById("grading").value;
    document.querySelectorAll("#courses tr").forEach(r => {
        const u = Math.abs(parseFloat(r.querySelector(".c-unit").value))||0;
        const g = r.querySelector(".c-grade").value;
        const gp = {A:scale==5?5:4,B:scale==5?4:3,C:scale==5?3:2,D:scale==5?2:1,E:1,F:0}[g]||0;
        units += u; pts += u * gp;
    });
    document.getElementById("gpa").innerText = units ? (pts/units).toFixed(2) : "0.00";
}

    let details = [];

        if (cgpa >= 3.5) return "Second Class Upper";
        if (cgpa >= 2.4) return "Second Class Lower";
        if (cgpa >= 1.5) return "Third Class";
        return "Pass";
    } else {

        if (cgpa >= 3.0) return "Second Class Upper";
        if (cgpa >= 2.0) return "Second Class Lower";
        if (cgpa >= 1.0) return "Third Class";
        return "Pass";
    }

function startExamTimer() {
    // Clear any existing timer
    if (examTimerInterval) {
        clearInterval(examTimerInterval);
    }
    
    // Start new timer that updates every second
    examTimerInterval = setInterval(() => {
        updateExamCountdowns();
    }, 1000);
    
    // Initial update
    updateExamCountdowns();
}

function updateExamCountdowns() {
    const now = new Date().getTime();
    exams.forEach((ex, idx) => {
        const examDateTime = new Date(ex.date + 'T' + ex.start).getTime();
        const diff = examDateTime - now;

        const cell = document.getElementById('t_' + idx);
        if (cell) {
            cell.innerHTML = getTimerString(diff);
        }

        // Check for 3-hour notification
        if (diff > 0 && diff <= (3 * 3600000) && !sentAlerts[ex.code + ex.date]) {
            if (Notification.permission === "granted") {
                new Notification("EXAM STARTING SOON", { 
                    body: `${ex.code} starts in 3 hours at ${ex.venue || 'Venue'}.`, 
                    requireInteraction: true 
                });
                sentAlerts[ex.code + ex.date] = true;
                localStorage.setItem('azura_sent_alerts', JSON.stringify(sentAlerts));
            }
        }
    });
}

function getTimerString(diff) {
    if (diff <= 0) return '<span style="color:#94a3b8; font-weight:normal;">Started/Past</span>';
    const d = Math.floor(diff / 86400000);
    const h = Math.floor((diff % 86400000) / 3600000);
    const m = Math.floor((diff % 3600000) / 60000);
    const s = Math.floor((diff % 60000) / 1000);
    return `${d}d ${h}h ${m}m ${s}s`;
}

function formatTo12Hr(timeStr) {
    if (!timeStr) return "";
    let [hours, minutes] = timeStr.split(':');
    let ampm = hours >= 12 ? 'PM' : 'AM';
    hours = hours % 12 || 12;
    return `${hours}:${minutes}${ampm}`;

function addLecture() {
    const title = document.getElementById("titleLecture").value;
    const code = document.getElementById("codeLecture").value.toUpperCase();
    const day = document.getElementById("day").value;
    const start = document.getElementById("startLecture").value;
    const venue = document.getElementById("venueLecture").value;

    if (!title || !code || !start) {
        showCustomNotification('Missing Information', 'Please fill in the course title, code, and start time.', 'warning');
        return;
    }

    const clash = lectureData.some(l => 
        l.day === day && l.start === start
    );

    if (clash) {
        showCustomNotification('Time Conflict', `This time slot already has a lecture on ${day}.`, 'warning');
        return;
    }

    const colors = ['#4338ca', '#1e293b', '#059669', '#7c3aed', '#db2777', '#d97706'];
    const color = colors[Math.floor(Math.random() * colors.length)];
    
    lectureData.push({ id: Date.now(), title, code, day, start, venue, color });
    localStorage.setItem('tt_lectures', JSON.stringify(lectureData));
    renderTimetable();
    showToast();
    showCustomNotification('Lecture Added', `${code} has been added to your timetable.`, 'success');
    ['titleLecture', 'codeLecture', 'startLecture', 'venueLecture'].forEach(id => document.getElementById(id).value = "");
}

function renderTimetable() {
    ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday'].forEach(d => document.getElementById(d).innerHTML = "");
    lectureData.sort((a,b) => a.start.localeCompare(b.start));

    lectureData.forEach(l => {
        const block = document.createElement("div");
        block.className = "course-block";
        block.style.background = l.color;
        block.innerHTML = `
            <span class="del-btn no-print" onclick="deleteLecture(${l.id})"><i class="fa-solid fa-circle-xmark"></i></span>
            <div style="font-weight:800; font-size: 13px; margin-bottom: 2px; text-transform:uppercase;">${l.code}</div>
            <div style="margin-bottom: 8px; font-weight: 300; opacity: 0.9;">${l.title}</div>
            <div style="font-size: 11px; font-weight: 700; display:flex; align-items:center; gap:5px;">
                <i class="fa-solid fa-clock"></i> ${formatTo12Hr(l.start)}
            </div>
            ${l.venue ? `<div style="font-size: 10px; opacity: 0.9; margin-top: 5px; display:flex; align-items:center; gap:5px;">
                <i class="fa-solid fa-location-dot"></i> ${l.venue}
            </div>` : ''}
        `;
        document.getElementById(l.day).appendChild(block);
    });
}

function switchTab(e, tabId) {
    // Update tabs
    document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
    document.querySelectorAll('.tab-link').forEach(l => l.classList.remove('active'));
    
    document.getElementById(tabId).classList.add('active');
    if (e) {
        e.currentTarget.classList.add('active');
    } else {
        // Find and activate the corresponding tab button
        const tabButtons = document.querySelectorAll('.tab-link');
        tabButtons.forEach(btn => {
            if (btn.onclick && btn.onclick.toString().includes(tabId)) {
                btn.classList.add('active');
            }
        });
    }
    
    // Save active tab to state
    appState.studyTab = tabId;
    saveAppState();
    
    // Load quotes when timer tab becomes active
    if (tabId === 'timerTab') {
        loadQuotes();
    }
}

/* --- PERSISTENT TIMER LOGIC --- */
function startTimerStudy() { 
    if(!timerRef) {
        isTimerPaused = false;
        timerRef = setInterval(() => { 
            secondsLeft--; 
            updateTimerUI();
            
            // Save state to local storage
            localStorage.setItem('az_timer_seconds', secondsLeft);
            localStorage.setItem('az_timer_break', isBreak);

            if(secondsLeft <= 0) { 
                bell.play(); 
                clearInterval(timerRef); 
                timerRef = null; 
                isBreak = !isBreak; 
                secondsLeft = isBreak ? 300 : 1500; 
                updatePauseResumeButton();
                startTimerStudy(); 
            } 
        }, 1000); 
        updatePauseResumeButton();
    }
}

function pauseTimer() { 
    if (timerRef && !isTimerPaused) {
        clearInterval(timerRef); 
        timerRef = null;
        isTimerPaused = true;
        updatePauseResumeButton();
    } else if (isTimerPaused) {
        startTimerStudy();
    }
}

function resetTimer() { 
    clearInterval(timerRef); 
    timerRef = null; 
    isBreak = false;
    secondsLeft = 1500; 
    isTimerPaused = false;
    localStorage.removeItem('az_timer_seconds');
    localStorage.removeItem('az_timer_break');
    updateTimerUI(); 
    updatePauseResumeButton();
}

function updateTimerUI() {
    const m = Math.floor(secondsLeft / 60), s = secondsLeft % 60;
    document.getElementById('timerDisplay').innerText = `${m}:${s < 10 ? '0' : ''}${s}`;
    document.getElementById('timerPhase').innerText = isBreak ? "BREAK PHASE" : "POMODORO SESSION";
}

function updatePauseResumeButton() {
    const pauseResumeBtn = document.getElementById('pauseResumeBtn');
    if (isTimerPaused) {
        pauseResumeBtn.innerHTML = '<i class="fa-solid fa-play"></i> Resume';
        pauseResumeBtn.style.background = 'var(--success)';
    } else {
        pauseResumeBtn.innerHTML = '<i class="fa-solid fa-pause"></i> Pause';
        pauseResumeBtn.style.background = 'var(--warning)';
    }
}

function saveToHistory() {
    const code = document.getElementById('recCode').value;
    const title = document.getElementById('recTitle').value;
    const topic = document.getElementById('recTopic').value;
    const body = document.getElementById('brainDump').value;
    if(!body.trim()) {
        showCustomNotification('Empty Work', 'Write something before saving.', 'warning');
        return;
    }

    history.unshift({
        id: Date.now(),
        code: code || "Untitled",
        title: title || "",
        topic: topic || "General Recall",
        body: body,
        date: new Date().toLocaleString()
    });
    localStorage.setItem('az_v14_history', JSON.stringify(history));
    renderHistory();
    showCustomNotification('Saved', 'Work added to History Vault.', 'success');
}

function renderHistory() {
    const list = document.getElementById('historyList');
    list.innerHTML = history.length ? '' : '<p style="font-size:12px; color:var(--text-muted)">No saved work found.</p>';
    history.forEach(item => {
        const div = document.createElement('div');
        div.className = 'history-item';
        div.innerHTML = `
            <div class="history-info">
                <b>${item.code} - ${item.topic}</b>
                <span>${item.date}</span>
            </div>
            <div style="display:flex; gap:5px;">
                <button class="btn btn-ghost btn-sm" onclick="loadHistoryItem(${item.id})"><i class="fa-solid fa-upload"></i></button>
                <button class="btn btn-sm" style="color:var(--danger)" onclick="deleteHistoryItem(${item.id})"><i class="fa-solid fa-trash"></i></button>
            </div>
        `;
        list.appendChild(div);
    });
}

function deleteHistoryItem(id) {

        }
    );
}

function addCourseStudy() {
    const title = document.getElementById('courseTitleStudy').value;
    const code = document.getElementById('courseCodeStudy').value.toUpperCase();
    const date = document.getElementById('examDateStudy').value;
    const days = Array.from(document.querySelectorAll('.day-check:checked')).map(i => i.value);
    
    if(!title || !code || !date || !days.length) {
        showCustomNotification('Missing Information', 'Please fill in course title, code, exam date, and select at least one study day.', 'warning');
        return;
    }
    
    coursesStudy.push({
        id: Date.now(), 
        title: title,
        code: code, 
        date: date,
        time: document.getElementById('studyTime').value, 
        hours: document.getElementById('dailyHours').value,
        days: days, 
        completed: false, 
        color: `hsl(${Math.random()*360}, 60%, 45%)`
    });
    saveStudy();
    
    // Clear form
    document.getElementById('courseCodeStudy').value = '';
    document.getElementById('examDateStudy').value = '';
    document.getElementById('courseTitleStudy').value = '';
    
    showCustomNotification('Plan Generated', `${code} study plan has been created.`, 'success');
}

function renderPlanner() {
    const body = document.getElementById('plannerBody'); 
    body.innerHTML = '';
    const weekdays = ['Monday','Tuesday','Wednesday','Thursday','Friday','Saturday','Sunday'];
    weekdays.forEach(d => { 
        body.innerHTML += `<tr><td><strong>${d}</strong></td><td id="cell-${d}"></td></tr>`; 
    });
    
    coursesStudy.forEach(c => {
        const diff = Math.ceil((new Date(c.date) - new Date()) / 86400000);
        c.days.forEach(day => {
            const cell = document.getElementById('cell-' + day);
            if(!cell) return;
            const card = document.createElement('div');
            card.className = `study-card ${c.completed ? 'done' : ''}`;
            card.style.background = c.color; 
            card.onclick = () => { 
                c.completed = !c.completed; 
                saveStudy(); 
                renderPlanner();
            };
            card.innerHTML = `
                <strong>${c.code}</strong>
                <span class="card-meta"><i class="fa-regular fa-clock"></i> ${c.time} | ${diff}d left</span>
                <button onclick="event.stopPropagation(); removeDay(${c.id}, '${day}')" style="position:absolute; bottom:8px; right:8px; border:none; background:none; color:white; cursor:pointer;">
                    <i class="fa-solid fa-circle-xmark"></i>
                </button>
            `;
            cell.appendChild(card);
        });
    });
}

function saveStudy() { 
    localStorage.setItem('az_v14_data', JSON.stringify(coursesStudy)); 
    renderPlanner(); 
}

function removeDay(id, dayName) {
    showStudyConfirmModal(
        'Remove Session?',
        'Are you sure you want to remove this study session?',
        () => {
            const idx = coursesStudy.findIndex(c => c.id === id);
            if(idx !== -1) { 
                coursesStudy[idx].days = coursesStudy[idx].days.filter(d => d !== dayName); 
                if(coursesStudy[idx].days.length === 0) coursesStudy.splice(idx, 1); 
                saveStudy(); 
                showCustomNotification('Session Removed', 'Study session has been removed.', 'success');
            }
        }
    );
}

function triggerPanicMode() {
    const today = new Intl.DateTimeFormat('en-US', { weekday: 'long' }).format(new Date());
    const tomorrow = new Intl.DateTimeFormat('en-US', { weekday: 'long' }).format(new Date(Date.now() + 86400000));
    let moved = 0;
    coursesStudy.forEach(c => { 
        if(c.days.includes(today) && !c.completed) { 
            if(!c.days.includes(tomorrow)) { 
                c.days.push(tomorrow); 
                moved++; 
            } 
        } 
    });
    saveStudy(); 
    showCustomNotification('Panic Mode Activated', `${moved} study sessions have been moved to tomorrow.`, 'warning');
}

async function loadQuotes() {
    try {
        const r = await fetch("https://raw.githubusercontent.com/ola1de/random-quotes/main/Quotes.txt");
        const t = await r.text(); 
        const q = t.split('\n').filter(l => l.includes("Quote:")).map(l => l.replace("Quote:", "").trim());
        if(q.length > 0) cachedQuotes = q;
    } catch(e) { 
        console.log("Using default quotes.");
    }
    
    // Initial display
    updateQuoteUI();
    
    // Set 10-second interval for real-time changes
    setInterval(updateQuoteUI, 10000);
}
