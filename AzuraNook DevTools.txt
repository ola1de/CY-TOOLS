    // URL ENCODING/DECODING FUNCTIONS
    // ============================
    function encodeUrl(url) {
        // Simple base64 encoding with URL safety
        return btoa(encodeURIComponent(url)).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }

    function decodeUrl(encoded) {
        try {
            // Add padding if needed
            while (encoded.length % 4) {
                encoded += '=';
            }
            encoded = encoded.replace(/-/g, '+').replace(/_/g, '/');
            return decodeURIComponent(atob(encoded));
        } catch (e) {
            return null;
        }
    }

    function initShortenerStats() {
        const today = new Date().toDateString();
        
        // Reset today's count if it's a new day
        if (shortenerStats.lastDate !== today) {

            saveShortenerStats();

        // Update UI
        if (document.getElementById('statsTotalNew')) {

            document.getElementById('statsTodayNew').textContent = shortenerStats.today;

    function checkForRedirect() {
        const urlParams = new URLSearchParams(window.location.search);
        const urlParam = urlParams.get('u');
        const timeParam = urlParams.get('t');
        
        if (urlParam) {
            // We're on a redirect page
            isOnRedirectPage = true;
            showRedirectPage();
            
            // Decode the URL
            const decodedUrl = decodeUrl(urlParam);
            
            if (decodedUrl && isValidUrl(decodedUrl)) {
                currentRedirectUrl = decodedUrl;
                currentRedirectTime = timeParam ? parseInt(timeParam) : 10;
                
                // Update stats
                shortenerStats.clicks++;
                saveShortenerStats();
                initShortenerStats();
                
                // Start the redirect countdown
                startRedirectCountdown();
            } else {
                showInvalidLink();
            }
        } else {
            // We're on the normal page
            isOnRedirectPage = false;
        }
    }

    function startRedirectCountdown() {
        let countdown = currentRedirectTime;
        const countdownNumber = document.getElementById('redirectCountdownNumber');
        const countdownSeconds = document.getElementById('redirectCountdownSeconds');
        const progressBar = document.getElementById('redirectProgressBar');
        
        if (!countdownNumber || !countdownSeconds || !progressBar) return;
        
        // Initialize display
        countdownNumber.textContent = countdown;
        countdownSeconds.textContent = countdown;
        progressBar.style.width = '100%';
        
        // Show the countdown box
        const countdownBox = document.getElementById('redirectCountdownBox');
        if (countdownBox) {
            countdownBox.style.display = 'block';
        }
        
        // Start countdown
        redirectInterval = setInterval(() => {
            countdown--;
            
            // Update display
            countdownNumber.textContent = countdown;
            countdownSeconds.textContent = countdown;
            
            // Update progress bar
            const progress = (countdown / currentRedirectTime) * 100;
            progressBar.style.width = progress + '%';
            
            // Redirect when countdown reaches 0
            if (countdown <= 0) {
                clearInterval(redirectInterval);
                performRedirect();
            }
        }, 1000);
    }

    function generateShortCode(length = 6) {
        const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
        let result = '';
        for (let i = 0; i < length; i++) {
            result += chars.charAt(Math.floor(Math.random() * chars.length));
        }
        return result;
    }

    function shortenUrlNew() {
        const originalUrl = document.getElementById('originalUrl')?.value.trim();
        const customSlug = document.getElementById('customSlug')?.value.trim();
        const countdownTime = parseInt(document.getElementById('countdownTime')?.value || 10);
        const enableEncryption = document.getElementById('enableEncryption')?.checked || true;
        
        // Validate URL
        if (!originalUrl) {
            showCustomNotification('Missing URL', 'Please enter a URL to shorten.', 'warning');
            return;
        }
        
        // Validate URL format
        if (!isValidUrl(originalUrl)) {
            showCustomNotification('Invalid URL', 'Please enter a valid URL (include http:// or https://).', 'warning');
            return;
        }
        
        // Generate short code
        let shortCode;
        if (customSlug) {
            // Clean custom slug
            shortCode = customSlug.replace(/[^a-zA-Z0-9-_]/g, '').toLowerCase();
            if (!shortCode) {
                showCustomNotification('Invalid Slug', 'Custom slug can only contain letters, numbers, hyphens, and underscores.', 'warning');
                return;
            }
        } else {
            // Generate random short code
            shortCode = generateShortCode();
        }
        
        // Encode the original URL
        const encodedUrl = enableEncryption ? encodeUrl(originalUrl) : encodeURIComponent(originalUrl);
        
        // Create short URL with parameters
        const baseUrl = window.location.origin + window.location.pathname;
        const shortUrl = `${baseUrl}?u=${encodedUrl}&t=${countdownTime}${customSlug ? `&c=${shortCode}` : ''}`;
        
        // Create URL data object for history
        const urlData = {
            id: Date.now(),
            shortCode: shortCode,
            originalUrl: originalUrl,
            shortUrl: shortUrl,
            countdownTime: countdownTime,
            clicks: 0,
            created: new Date().toISOString(),
            createdDate: new Date().toDateString()
        };
        
        // Add to database

    function adjustColor(color, amount) {
        let usePound = false;
        if (color[0] === "#") {
            color = color.slice(1);
            usePound = true;
        }
        const num = parseInt(color, 16);
        let r = (num >> 16) + amount;
        if (r > 255) r = 255;
        else if (r < 0) r = 0;
        let b = ((num >> 8) & 0x00FF) + amount;
        if (b > 255) b = 255;
        else if (b < 0) b = 0;
        let g = (num & 0x0000FF) + amount;
        if (g > 255) g = 255;
        else if (g < 0) g = 0;
        return (usePound ? "#" : "") + (g | (b << 8) | (r << 16)).toString(16).padStart(6, '0');
    }

    function processEncoder() {
        const input = document.getElementById('encoderInput').value;
        const type = document.getElementById('encodeType').value;
        let output = '';
        
        try {
            switch(type) {
                case 'urlEncode':
                    output = encodeURIComponent(input);
                    break;
                case 'urlDecode':
                    output = decodeURIComponent(input);
                    break;
                case 'base64Encode':
                    output = btoa(unescape(encodeURIComponent(input)));
                    break;
                case 'base64Decode':
                    output = decodeURIComponent(escape(atob(input)));
                    break;
                case 'htmlEncode':
                    output = input.replace(/[&<>"']/g, m => ({
                        '&': '&amp;',
                        '<': '&lt;',
                        '>': '&gt;',
                        '"': '&quot;',
                        "'": '&#39;'
                    }[m]));
                    break;
                case 'htmlDecode':
                    const textarea = document.createElement('textarea');
                    textarea.innerHTML = input;
                    output = textarea.value;
                    break;
            }
            
            document.getElementById('encoderOutput').textContent = output;
            document.getElementById('encoderResult').classList.add('active');
            showNotification('Success', 'Text processed successfully!', 'success');
            
        } catch (e) {
            showNotification('Error', 'Invalid input for selected operation.', 'error');
        }
    }

    function hexToRGB(hex) {
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return { r, g, b };
    }
    
    function RGBToHSL(r, g, b) {
        r /= 255;
        g /= 255;
        b /= 255;
        
        const max = Math.max(r, g, b);
        const min = Math.min(r, g, b);
        let h, s, l = (max + min) / 2;
        
        if (max === min) {
            h = s = 0;
        } else {
            const d = max - min;
            s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
            switch(max) {
                case r: h = (g - b) / d + (g < b ? 6 : 0); break;
                case g: h = (b - r) / d + 2; break;
                case b: h = (r - g) / d + 4; break;
            }
            h /= 6;
        }
        
        return {
            h: Math.round(h * 360),
            s: Math.round(s * 100),
            l: Math.round(l * 100)
        };
    }
    
    function HSLToHex(h, s, l) {
        s /= 100;
        l /= 100;
        
        const c = (1 - Math.abs(2 * l - 1)) * s;
        const x = c * (1 - Math.abs((h / 60) % 2 - 1));
        const m = l - c / 2;
        
        let r, g, b;
        
        if (h >= 0 && h < 60) {
            [r, g, b] = [c, x, 0];
        } else if (h >= 60 && h < 120) {
            [r, g, b] = [x, c, 0];
        } else if (h >= 120 && h < 180) {
            [r, g, b] = [0, c, x];
        } else if (h >= 180 && h < 240) {
            [r, g, b] = [0, x, c];
        } else if (h >= 240 && h < 300) {
            [r, g, b] = [x, 0, c];
        } else {
            [r, g, b] = [c, 0, x];
        }
        
        r = Math.round((r + m) * 255);
        g = Math.round((g + m) * 255);
        b = Math.round((b + m) * 255);
        
        return `#${((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1)}`;
    }
    
    function generateColorScheme() {
        const hex = document.getElementById('hexColor').value;
        const type = document.getElementById('schemeType').value;
        const rgb = hexToRGB(hex);
        const hsl = RGBToHSL(rgb.r, rgb.g, rgb.b);
        
        let colors = [];
        
        switch(type) {
            case 'monochromatic':
                for (let i = -2; i <= 2; i++) {
                    const newHsl = { ...hsl, l: Math.max(0, Math.min(100, hsl.l + (i * 15))) };
                    colors.push(HSLToHex(hsl.h, hsl.s, newHsl.l));
                }
                break;
            case 'analogous':
                for (let i = -2; i <= 2; i++) {
                    const newHsl = { ...hsl, h: (hsl.h + (i * 30) + 360) % 360 };
                    colors.push(HSLToHex(newHsl.h, hsl.s, hsl.l));
                }
                break;
            case 'complementary':
                colors = [hex, HSLToHex((hsl.h + 180) % 360, hsl.s, hsl.l)];
                colors.push(HSLToHex(hsl.h, hsl.s, Math.min(100, hsl.l + 20)));
                colors.push(HSLToHex(hsl.h, hsl.s, Math.max(0, hsl.l - 20)));
                colors.push(HSLToHex((hsl.h + 180) % 360, hsl.s, Math.min(100, hsl.l + 20)));
                break;
            case 'triadic':
                colors = [
                    hex,
                    HSLToHex((hsl.h + 120) % 360, hsl.s, hsl.l),
                    HSLToHex((hsl.h + 240) % 360, hsl.s, hsl.l)
                ];
                colors.push(HSLToHex(hsl.h, Math.min(100, hsl.s + 20), hsl.l));
                colors.push(HSLToHex((hsl.h + 120) % 360, Math.min(100, hsl.s + 20), hsl.l));
                break;
            case 'tetradic':
                colors = [
                    hex,
                    HSLToHex((hsl.h + 90) % 360, hsl.s, hsl.l),
                    HSLToHex((hsl.h + 180) % 360, hsl.s, hsl.l),
                    HSLToHex((hsl.h + 270) % 360, hsl.s, hsl.l)
                ];
                colors.push(HSLToHex(hsl.h, hsl.s, Math.min(100, hsl.l + 20)));
                break;
        }
        
        const container = document.getElementById('colorSchemePreview');
        container.innerHTML = colors.map(color => `
            <div style="text-align: center;">

    function updateTickerPreview() {
